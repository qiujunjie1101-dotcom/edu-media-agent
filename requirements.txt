# ---------- 工作流编排 ----------
# LangGraph 1.x：提供 StateGraph（状态机）、interrupt/Command（人工中断与恢复）、
# 以及 InMemorySaver / PostgresSaver（检查点持久化）。
# 这里锁定具体版本，因为 1.0 前后部分 API 的导入路径与语义有过调整，
# 升级前请先跑通 tests/integration/test_workflow.py。
langgraph==1.2.11

# ---------- 检查点持久化（S5） ----------
# LangGraph 的 PostgreSQL 检查点实现。**不是** langgraph 的默认依赖，
# 必须单独安装；且它只在 CHECKPOINTER_BACKEND=postgres 时才会被导入，
# 因此 memory 模式（本地开发 / 绝大多数测试）不会用到它。
langgraph-checkpoint-postgres>=3.1.0

# PostgreSQL 驱动与连接池。AsyncPostgresSaver 通过它读写检查点：
#   [binary] 装预编译的 C 扩展，省去本地编译（macOS/Linux 都有轮子）
#   [pool]   带上 psycopg_pool，连接池在它里面
# 没有它，postgres 模式会在创建 Checkpointer 时报 ImportError。
psycopg[binary,pool]>=3.2.0

# ---------- Web 框架与 ASGI 服务器 ----------
fastapi>=0.115.0            # 提供 HTTP 路由、依赖注入、自动生成 OpenAPI 文档
uvicorn[standard]>=0.30.0   # ASGI 服务器，负责真正监听端口并把请求交给 FastAPI

# ---------- 数据校验与配置 ----------
pydantic>=2.7.0             # v2 版本，负责把 dict/JSON 校验成强类型对象（领域模型的基础）
pydantic-settings>=2.2.0    # 让配置类可以从环境变量 / .env 文件读取值
python-dotenv>=1.0.0        # pydantic-settings 读取 .env 文件时依赖的解析库

# ---------- HTTP 客户端（测试用） ----------
httpx>=0.27.0               # 异步 HTTP 客户端；测试 FastAPI 时用它直接调用 ASGI 应用，无需真启端口

# ---------- 测试框架 ----------
pytest>=8.0.0               # 测试运行器
pytest-asyncio>=0.23.0      # 让 pytest 能直接运行 async def 测试函数
